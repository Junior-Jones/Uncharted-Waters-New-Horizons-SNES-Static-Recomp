cmake_minimum_required(VERSION 3.28)
# Compiler identification compiles an object, and ABI probes build a static
# library. Neither requires creating a compiler-test executable.
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(UnchartedWatersStaticCore LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/static-core/generated/scpu/nh_v07_sources.txt" NH_SOURCE_NAMES)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/static-core/generated/scpu/V07C-generation-manifest.json" NH_GENERATION)
string(JSON NH_CONTEXTS GET "${NH_GENERATION}" context_count)
if(NOT NH_CONTEXTS EQUAL 283586)
  message(FATAL_ERROR "The complete Version07 exact context manifest changed")
endif()
# Reuse the admitted CPU universe, replacing only the qualified executable-
# WRAM copy body. One current dispatcher remains; no historical runtime layer.
list(REMOVE_ITEM NH_SOURCE_NAMES "static-core/generated/scpu/nh_v07_shard_007EF5.c")
set(NH_COPY_SUCCESSOR "static-core/candidate/block-copy-knownness/generated/scpu")
list(APPEND NH_SOURCE_NAMES "${NH_COPY_SUCCESSOR}/nh_v07_shard_007EF5.c")
foreach(NH_COPY_SHARD 00C094 00C096 00C0A3 00C1F8)
  list(REMOVE_ITEM NH_SOURCE_NAMES "static-core/generated/scpu/nh_v07_shard_${NH_COPY_SHARD}.c")
  list(APPEND NH_SOURCE_NAMES "${NH_COPY_SUCCESSOR}/nh_v07_shard_${NH_COPY_SHARD}.c")
endforeach()
set(NH_OWNER_NAMES bus_map rom_identity memory fetch bus_runtime math_io cpu_io
                   sram_file clock_domain clock_port machine scheduler scheduler_io
                   autojoy timing_plan transfer_timing dma ppu renderer apu spc700 input sdsp core)
foreach(NH_OWNER IN LISTS NH_OWNER_NAMES)
  list(APPEND NH_SOURCE_NAMES "static-core/src/new_horizons_${NH_OWNER}.c")
endforeach()
list(APPEND NH_SOURCE_NAMES "static-core/generated/timing/new_horizons_timing_data.c")
list(APPEND NH_SOURCE_NAMES "static-core/generated/spc700/new_horizons_spc700_aot.c")
set(NH_SOURCES)
foreach(NH_NAME IN LISTS NH_SOURCE_NAMES)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${NH_NAME}" OR NH_NAME MATCHES "(^|/)\\.\\.(/|$)")
    message(FATAL_ERROR "Invalid current source allow-list entry: ${NH_NAME}")
  endif()
  list(APPEND NH_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/${NH_NAME}")
endforeach()
list(LENGTH NH_SOURCES NH_SOURCE_COUNT)
list(REMOVE_DUPLICATES NH_SOURCES)
list(LENGTH NH_SOURCES NH_UNIQUE_COUNT)
if(NOT NH_SOURCE_COUNT EQUAL NH_UNIQUE_COUNT OR NOT NH_SOURCE_COUNT EQUAL 2573)
  message(FATAL_ERROR "Duplicate or incomplete current native source universe")
endif()

set(NH_COMPACT_SOURCE_ROOT "" CACHE PATH "Opt-in verified V18C generated-source overlay")
set(NH_SHARED_SOURCE_ROOT "" CACHE PATH "Opt-in V18C exact-key shared native stage")
option(NH_SIZE_OPTIMIZED_GENERATED "Size optimization for generated authority only" OFF)
if(NH_COMPACT_SOURCE_ROOT AND NH_SHARED_SOURCE_ROOT)
  message(FATAL_ERROR "Select only one V18C representation")
endif()
set(NH_SELECTED_SOURCES)
set(NH_GENERATED_SOURCES)
foreach(NH_SOURCE IN LISTS NH_SOURCES)
  if(NH_SOURCE MATCHES "/nh_v07_shard_[0-9A-F]+\\.c$")
    if(NH_SHARED_SOURCE_ROOT)
      continue()
    endif()
    if(NH_COMPACT_SOURCE_ROOT)
      file(RELATIVE_PATH NH_RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${NH_SOURCE}")
      set(NH_SOURCE "${NH_COMPACT_SOURCE_ROOT}/${NH_RELATIVE}")
      if(NOT EXISTS "${NH_SOURCE}")
        message(FATAL_ERROR "Missing selected compacted owner: ${NH_SOURCE}")
      endif()
    endif()
    list(APPEND NH_GENERATED_SOURCES "${NH_SOURCE}")
  elseif(NH_SOURCE MATCHES "/static-core/generated/(timing|spc700)/")
    list(APPEND NH_GENERATED_SOURCES "${NH_SOURCE}")
  endif()
  list(APPEND NH_SELECTED_SOURCES "${NH_SOURCE}")
endforeach()
if(NH_SHARED_SOURCE_ROOT)
  file(READ "${NH_SHARED_SOURCE_ROOT}/shared-native-receipt.json" NH_SHARED_RECEIPT)
  string(JSON NH_SHARED_CONTEXTS GET "${NH_SHARED_RECEIPT}" context_count)
  string(JSON NH_SHARED_REMOVED GET "${NH_SHARED_RECEIPT}" contexts_removed)
  if(NOT NH_SHARED_CONTEXTS EQUAL 283586 OR NOT NH_SHARED_REMOVED EQUAL 0)
    message(FATAL_ERROR "Shared exact-context authority is incomplete")
  endif()
  file(STRINGS "${NH_SHARED_SOURCE_ROOT}/native/nh_v18_shared_sources.txt" NH_SHARED_NAMES)
  foreach(NH_SHARED_NAME IN LISTS NH_SHARED_NAMES)
    if(NOT NH_SHARED_NAME MATCHES "^native/nh_v18_shared_[A-Za-z0-9_]+\\.c$"
       OR NOT EXISTS "${NH_SHARED_SOURCE_ROOT}/${NH_SHARED_NAME}")
      message(FATAL_ERROR "Invalid shared native source: ${NH_SHARED_NAME}")
    endif()
    list(APPEND NH_SELECTED_SOURCES "${NH_SHARED_SOURCE_ROOT}/${NH_SHARED_NAME}")
    list(APPEND NH_GENERATED_SOURCES "${NH_SHARED_SOURCE_ROOT}/${NH_SHARED_NAME}")
  endforeach()
endif()
add_library(uncharted_waters_static_core STATIC ${NH_SELECTED_SOURCES})
if(NH_SIZE_OPTIMIZED_GENERATED)
  if(MSVC)
    set_property(SOURCE ${NH_GENERATED_SOURCES} APPEND PROPERTY COMPILE_OPTIONS /O1)
  else()
    set_property(SOURCE ${NH_GENERATED_SOURCES} APPEND PROPERTY COMPILE_OPTIONS -Os)
  endif()
endif()
target_include_directories(uncharted_waters_static_core PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}/static-core/include"
  "${CMAKE_CURRENT_SOURCE_DIR}/static-core/internal"
  "${CMAKE_CURRENT_SOURCE_DIR}/static-core/generated/scpu"
  "${CMAKE_CURRENT_SOURCE_DIR}/static-core/generated/spc700")
if(MSVC)
  target_compile_options(uncharted_waters_static_core PRIVATE /O2 /W4 /WX /bigobj)
else()
  target_compile_options(uncharted_waters_static_core PRIVATE -O2 -Wall -Wextra -Werror)
endif()
set_property(GLOBAL PROPERTY JOB_POOLS nh_native_compile=2)
set_property(TARGET uncharted_waters_static_core PROPERTY JOB_POOL_COMPILE nh_native_compile)

option(NH_BUILD_HEADLESS_CONNECTOR "Build the Version17 target-owned connector" OFF)
if(NH_BUILD_HEADLESS_CONNECTOR)
  add_executable(new_horizons_headless tools/new_horizons_headless.c)
  target_link_libraries(new_horizons_headless PRIVATE uncharted_waters_static_core)
  set_target_properties(new_horizons_headless PROPERTIES OUTPUT_NAME "NewHorizonsHeadless")
  if(MSVC)
    target_compile_options(new_horizons_headless PRIVATE /O2 /W4 /WX)
  else()
    target_compile_options(new_horizons_headless PRIVATE -O2 -Wall -Wextra -Werror)
  endif()
endif()
# The default selection creates no application, frontend, native test
# executable or ROM input. Version17 opts into its target-owned connector.
option(NH_BUILD_WINDOWS_FRONTEND "Build the ported Windows Launcher" OFF)
if(NH_BUILD_WINDOWS_FRONTEND)
  add_subdirectory(frontend/frontend/windows)
endif()
